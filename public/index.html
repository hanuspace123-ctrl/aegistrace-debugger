Telemetry Ingestion Address: Active HTTP Listener

Code Integration Example (Node.js / Express):

const { OTLPTraceExporter } = require('@opentelemetry/exporter-trace-otlp-http');
const { NodeTracerProvider } = require('@opentelemetry/sdk-trace-node');

const exporter = new OTLPTraceExporter({
  url: 'https://ecommerce-visual-debugger.vercel.app/api/telemetry/otlp',
  headers: { 'Authorization': 'Bearer YOUR_STORE_API_TOKEN' }
});

const provider = new NodeTracerProvider();
provider.addSpanProcessor(new SimpleSpanProcessor(exporter));
provider.register();
Send External Alert or Order Event via cURL: 
curl -X POST https://ecommerce-visual-debugger.vercel.app/api/telemetry/webhook \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_STORE_TOKEN" \
  -d '{"traceId":"trace_prod_9981","service":"payment","status":"failed","httpStatus":504,"latencyMs":2500,"errorSignature":"CONN_TIMEOUT_STRIPE_API"}'
